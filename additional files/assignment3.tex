\section{MVP v1 Access Instructions}

\begin{enumerate}[leftmargin=*]
    \item Clone the repository at the submission commit:
\begin{verbatim}
git clone https://github.com/Fedos113/SWP_TickFrame_28_team.git
cd SWP_TickFrame_28_team
git checkout f1a0d72bfde44a2b1e323cd3588f40a50e8dad23
\end{verbatim}

    \item Copy the environment template and fill in credentials:
\begin{verbatim}
cp .env.example .env
# Edit .env — set BYBIT_API_KEY and BYBIT_API_SECRET (read-only testnet keys)
\end{verbatim}

    \item Install dependencies (Python 3.10+):
\begin{verbatim}
pip install -r requirements.txt
\end{verbatim}

    \item Run the CLI or start the web dashboard:
\begin{verbatim}
python -m tickframe scan --symbol BTCUSDT --interval 1h --limit 10
python -m tickframe serve --port 5000
\end{verbatim}

    \item Open \texttt{http://localhost:5000} in a browser to access the interactive chart dashboard.
\end{enumerate}
